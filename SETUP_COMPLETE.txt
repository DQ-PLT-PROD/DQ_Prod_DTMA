📊 DATABASE SCHEMA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

learner_profiles table:
  • id (UUID) - User ID
  • email (TEXT) - User email
  • full_name (TEXT) - Display name
  • current_stage (INTEGER) - Current stage (0-10)
  • completed_stages (INTEGER[]) - Completed stages array
  • total_score (INTEGER) - Total points
  • badges (TEXT[]) - Badge names array
  • created_at (TIMESTAMP) - Account creation
  • updated_at (TIMESTAMP) - Last update
